printf format string

%u : 부호없는 10진수 int
%04x : 0을 채워서, 4자리, 16진수, 소문자
%016I64X : 0을 채워서, 16자리, int64, 16진수, 대문자
%I64d: int64, 부호있는 10진수 int

// 64bit = [18]0xffffffffffffffff = [20]18446744073709551615
// 32bit = [10]0xffffffff         = [10]4294967295

//===========================================================================
template <typename T=std::string>
class sprintf_converter
{
public:
    template <int size, typename T_FORMAT, typename T_VALUE>
    static T convert (T_FORMAT format, T_VALUE v)
    {
        char buf[size];   

#pragma warning( push )
#pragma warning( disable : 4996 )
        sprintf (buf, format, v);
#pragma warning( pop )

        T output(buf);
        return output;
    }
};

//===========================================================================
template <typename T=std::wstring>
class swprintf_converter
{
public:
    template <int size, typename T_VALUE>
    static T convert (const wchar_t* format, T_VALUE v)
    {
        wchar_t buf[size];   

#pragma warning( push )
#pragma warning( disable : 4996 )
        swprintf (buf, format, v);
#pragma warning( pop )

        T output(buf);
        return output;
    }
};

//===========================================================================
class sscanf_converter
{
public:
    template <typename T_BUFFER, typename T_FORMAT, typename T_VALUE>
    static T_VALUE convert (T_BUFFER buffer, T_FORMAT format, T_VALUE value)
    {
        if (buffer)
        {
#pragma warning( push )
#pragma warning( disable : 4996 )
            sscanf (buffer, format, &value);
#pragma warning( pop )
        }
        return value;
    }
};

//===========================================================================
class swscanf_converter
{
public:
    template <typename T_BUFFER, typename T_FORMAT, typename T_VALUE>
    static T_VALUE convert (T_BUFFER buffer, T_FORMAT format, T_VALUE value)
    {
        if (buffer)
        {
#pragma warning( push )
#pragma warning( disable : 4996 )
            swscanf (buffer, format, &value);
#pragma warning( pop )
        }

        return value;
    }
};

template
<
    typename                   T,
    template<typename T> class T_CONVERTER
>
class string_mbcs_maker : public T_CONVERTER<T>
{
public:
    static T sint      (signed   int     v){ return convert<16>("%d"     , v); }
    static T uint      (unsigned int     v){ return convert<16>("%u"     , v); }
    static T sint_64   (signed   __int64 v){ return convert<32>("%I64d"  , v); }
    static T uint_64   (unsigned __int64 v){ return convert<32>("%I64u"  , v); }
    static T hexa      (unsigned int     v){ return convert<16>("%x"     , v); }
    static T hexa_sh   (unsigned int     v){ return convert<16>("#%x"    , v); }
    static T hexa_0x   (unsigned int     v){ return convert<16>("0x%x"   , v); }
    static T hexa_64   (unsigned __int64 v){ return convert<32>("%I64x"  , v); }
    static T hexa_64sh (unsigned __int64 v){ return convert<32>("#%I64x" , v); }
    static T hexa_640x (unsigned __int64 v){ return convert<32>("0x%I64x", v); }
    static T HEXA      (unsigned int     v){ return convert<16>("%X"     , v); }
    static T HEXA_sh   (unsigned int     v){ return convert<16>("#%X"    , v); }
    static T HEXA_0x   (unsigned int     v){ return convert<16>("0x%X"   , v); }
    static T HEXA_64   (unsigned __int64 v){ return convert<32>("%I64X"  , v); }
    static T HEXA_64sh (unsigned __int64 v){ return convert<32>("#%I64X" , v); }
    static T HEXA_640x (unsigned __int64 v){ return convert<32>("0x%I64X", v); }
    static T real      (double           v){ return convert<64>("%.1f"   , v); }
};
